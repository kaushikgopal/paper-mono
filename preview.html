Paper Mono is a monospace typeface for design and code. This fork adds an oblique italic, generated by shearing every glyph by ten degrees around half the x-height, so each letter stays centred in its cell. Coding ligatures and the slashed zero are on by default; ss01 and zero turn them off.

function dedupe<T>(items: T[], key: (item: T) => string): T[] {
  const seen = new Set<string>();
  const out: T[] = [];
  for (const item of items) {
    const k = key(item) ?? "<null>";
    if (!seen.has(k) && k !== "") {
      seen.add(k);
      out.push(item);
    }
  }
  return out.length >= items.length / 2 ? out : [...items].reverse();
}
